  async function loadSourceText(source) {
    const sep = source.url.includes("?") ? "&" : "?";
    const req = new Request(`${source.url}${sep}_velvet=${Date.now()}-${Math.random()}`);
    req.timeoutInterval = 18;
    req.headers = {
      "User-Agent": `VelvetFeedSync/${VERSION}`,
      "Accept": "text/html,application/xhtml+xml",
      "Cache-Control": "no-cache",
      "Pragma": "no-cache",
      "Referer": source.site + "/"
    };
    return await req.loadString();
  }

  function baselineForCard(meta, sourceName, rank, diagnostics) {
    const profiles = {};
    const rejectionSeen = new Set();

    for (const [profileName, profile] of Object.entries(PROFILES)) {
      const sourceDef = profile.sources[sourceName];
      if (!sourceDef || rank > Number(sourceDef.maxRank || 100)) continue;
      const gate = profileGate(meta, sourceDef, profileName, profile.stimulusMax === true);
      if (!gate.allowed) {
        const key = `${gate.reason}`;
        if (!rejectionSeen.has(key)) {
          rejectionSeen.add(key);
          noteCount(diagnostics.baseline_rejections, gate.reason, sourceName);
          if (SAFETY_REASONS.has(gate.reason)) noteCount(diagnostics.quarantine, gate.reason, sourceName);
        }
        continue;
      }
      const rankScore = Math.max(0.04, (101 - Math.min(100, rank)) / 100) * 0.34;
      profiles[profileName] = {
        eligible: true,
        score: Number((0.22 + rankScore + gate.score).toFixed(5)),
        run_cap: Number(sourceDef.runCap || 3),
        max_rank: Number(sourceDef.maxRank || 100),
        // Compatibility: old clients only understand exposure_tier and must not
        // treat a low-confidence source prior as per-post evidence.
        exposure_tier: gate.exposureBasis === "metadata" ? gate.exposureTier : "unknown",
        exposure_estimate_tier: gate.exposureTier,
        exposure_basis: gate.exposureBasis || "unknown",
        exposure_confidence: Number.isFinite(Number(gate.exposureConfidence)) ? Number(gate.exposureConfidence) : 0,
        intensity: intensityFromTier(gate.exposureTier, gate.exposureBasis),
        post_jp: gate.postJpEvidence === true,
        post_female: gate.postFemaleEvidence === true,
        age_target: gate.ageTarget === true
      };
    }

    return profiles;
  }
