    });
  }

  function stripNegatedTerms(text, terms) {
    let t = textOf(text);
    for (const term of terms) {
      const raw = textOf(term).trim();
      if (!raw || !/^[a-z0-9 -]+$/.test(raw)) continue;
      const pattern = raw.split(/\s+/).map(escapeRegex).join("\\s+");
      t = t.replace(new RegExp(`\\b(?:no|not|without|exclude(?:d|s|ing)?|avoid(?:ed|s|ing)?|ban(?:ned|s|ning)?|prohibit(?:ed|s|ing)?)\\s+(?:any\\s+)?${pattern}\\b`, "gi"), " ");
    }
    return t;
  }

  function sanitizeContext(value) {
    let t = textOf(value);
    t = stripNegatedTerms(t, [...HARD_AGE_RISK, ...MALE_EXCLUDE, ...BODY_EXCLUDE, ...AGE_EXCLUDE]);
    const jpRisk = "(?:未成年|児童|女子高生|高校生|中学生|小学生|ロリ|ショタ|男性|男|おじさん|おっさん|肥満|ぽっちゃり|デブ|熟女|おばさん|お婆さん|老婆)";
    t = t.replace(new RegExp(`${jpRisk}\\s*(?:禁止|不可|ng|お断り|除外|なし|無し)`, "gi"), " ");
    t = t.replace(new RegExp(`(?:禁止|不可|ng|お断り|除外)\\s*(?:の|:|：)?\\s*${jpRisk}`, "gi"), " ");
    return t.replace(/\s+/g, " ").trim();
  }

  function explicitAge(text) {
    const t = textOf(text);
    const matches = [
      ...t.matchAll(/\b(\d{1,2})\s*(?:yo|y\/o|y\.o\.?|years?\s*old)\b/gi),
      ...t.matchAll(/\bage\s*(\d{1,2})\b/gi),
      ...t.matchAll(/(?:^|[^\d])(\d{1,2})\s*歳(?:[^\d]|$)/g)
    ];
    const ages = matches.map(m => Number(m[1])).filter(Number.isFinite);
    if (ages.some(age => age < 18)) return { under18: true, adult: false, age: Math.min(...ages) };
    const adults = ages.filter(age => age >= 18 && age <= 99);
    return adults.length ? { under18: false, adult: true, age: Math.max(...adults) } : { under18: false, adult: false, age: null };
  }

  function decodeHtml(value) {
    return String(value || "").replace(/&amp;/gi, "&").replace(/&quot;/gi, '"').replace(/&#39;|&apos;/gi, "'").replace(/&lt;/gi, "<").replace(/&gt;/gi, ">");
  }
  function stripHtml(value) {
    return decodeHtml(value).replace(/<script[\s\S]*?<\/script>/gi, " ").replace(/<style[\s\S]*?<\/style>/gi, " ").replace(/<[^>]+>/g, " ").replace(/\s+/g, " ").trim();
  }
  function decodeEscaped(value) {
    return decodeHtml(String(value || "")).replace(/\\\//g, "/").replace(/\\u0026/gi, "&").replace(/\\n/g, " ").replace(/\\"/g, '"');
  }
  function metadataText(block) {
    const pieces = [stripHtml(block)];
    for (const m of String(block).matchAll(/(?:alt|title|aria-label)\s*=\s*["']([^"']{1,500})["']/gi)) pieces.push(decodeHtml(m[1]));
    for (const m of String(block).matchAll(/"(?:full_text|description|caption|text)"\s*:\s*"([^"]{1,700})"/gi)) pieces.push(decodeEscaped(m[1]));
    return sanitizeContext(pieces.join(" "));
  }

  function profileGate(meta, sourceDef, profileName, stimulusMax) {
    const plain = sanitizeContext(meta);
    if (hasTerm(plain, NON_TARGET_CUES)) return { allowed: false, reason: "non-target" };
    if (hasTerm(plain, HARD_AGE_RISK)) return { allowed: false, reason: "hard-age-risk" };
    const age = explicitAge(plain);
    if (age.under18) return { allowed: false, reason: "numeric-under18" };
    if (hasTerm(plain, NONCONSENSUAL)) return { allowed: false, reason: "nonconsensual-metadata" };
    if (age.adult && age.age > 25) return { allowed: false, reason: `age-${age.age}` };
    if (hasTerm(plain, AGE_EXCLUDE)) return { allowed: false, reason: "age-profile" };
    if (hasTerm(plain, MALE_EXCLUDE)) return { allowed: false, reason: "male-profile" };
    if (hasTerm(plain, BODY_EXCLUDE)) return { allowed: false, reason: "body-profile" };

    const adultCue = hasTerm(plain, ADULT_CUES) || age.adult;
    if (!sourceDef.adultTrust && !adultCue) return { allowed: false, reason: "no-adult-evidence" };
    const jpDirect = hasTerm(plain, JP_DIRECT_CUES);
    const postJpEvidence = jpDirect || /[ぁ-んァ-ヶー]/.test(plain);
    const explicitNonJp = hasTerm(plain, NON_JP_CUES) || /[가-힣]/.test(plain);
    if (explicitNonJp && !jpDirect) return { allowed: false, reason: "non-jp" };
