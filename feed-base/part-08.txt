
  async function promptToken() {
    if (Keychain.contains(TOKEN_KEY)) return Keychain.get(TOKEN_KEY);
    const alert = new Alert();
    alert.title = "Velvet Feed Sync";
    alert.message = "GitHub fine-grained tokenを1回だけ保存します。対象repoは eros-hub、権限は Contents: Read and write だけでOK。";
    if (typeof alert.addSecureTextField === "function") alert.addSecureTextField("github_pat_...", "");
    else alert.addTextField("github_pat_...", "");
    alert.addAction("保存して続行");
    alert.addCancelAction("キャンセル");
    const choice = await alert.presentAlert();
    if (choice < 0) return null;
    const token = String(alert.textFieldValue(0) || "").trim();
    if (token.length < 20) throw new Error("GitHub tokenが短すぎます");
    Keychain.set(TOKEN_KEY, token);
    return token;
  }

  function githubHeaders(token) {
    return {
      "Authorization": `Bearer ${token}`,
      "Accept": "application/vnd.github+json",
      "X-GitHub-Api-Version": "2022-11-28",
      "User-Agent": `VelvetFeedSync/${VERSION}`,
      "Content-Type": "application/json"
    };
  }

  async function githubJson(url, token, method = "GET", body = null) {
    const req = new Request(url);
    req.method = method;
    req.timeoutInterval = 25;
    req.headers = githubHeaders(token);
    if (body) req.body = JSON.stringify(body);
    let raw;
    try {
      raw = await req.loadString();
    } catch (error) {
      const status = Number(req.response?.statusCode || 0);
      if (status === 401 || status === 403) {
        try { Keychain.remove(TOKEN_KEY); } catch (_) {}
        throw new Error(`GitHub認証失敗 (${status || "network"})。tokenを再設定してください。`);
      }
      throw error;
    }
    const status = Number(req.response?.statusCode || 0);
    let payload = null;
    try { payload = raw ? JSON.parse(raw) : null; } catch (_) {}
    if (status < 200 || status >= 300) {
      if (status === 401 || status === 403) {
        try { Keychain.remove(TOKEN_KEY); } catch (_) {}
      }
      throw new Error(`GitHub HTTP ${status}: ${String(payload?.message || raw || "unknown").slice(0, 180)}`);
    }
    return payload;
  }

  async function currentGitHubFile(token) {
    const url = `https://api.github.com/repos/${OWNER}/${REPO}/contents/${encodeURIComponent(TARGET_PATH)}?ref=${encodeURIComponent(TARGET_BRANCH)}`;
    return await githubJson(url, token);
  }

  function stableSignature(payload) {
    return JSON.stringify({
      canonical: payload?.canonical_baseline || null,
      items: (payload?.items || []).map(item => ({ id: item.id, image: item.image_url, baseline: item.baseline })),
      diagnostics: payload?.diagnostics || {},
      failures: payload?.failures || []
    });
  }
